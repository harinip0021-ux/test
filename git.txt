git commands

git --version -Shows the installed Git version
git config -Configures Git settings
git init -Creates a new Git repository
git status -Shows the current status of files
git add	<file> -Adds files to the staging area
git add . -Adds all changed files
git commit -m "message" -Saves changes with a message
git log -Shows previous commits
git clone<repro_url> -Copies a GitHub repository to your computer
git remote -v -Shows connected remote repositories
git remote add origin<URL> -Connects your local project to GitHub
git push origin<branch> -Uploads commits to GitHub
git pull origin<branch> -Downloads and merges changes from GitHub
git branch -Shows available branches
git branch name -Creates a new branches
git help -Shows Git help/documentation
git config --global user.email "example.com" - set global email
git config --global user.name "name" - set global name
git checkout<branch> - switch to another branchs
